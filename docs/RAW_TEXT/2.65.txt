(define (intersection-set set1 set2)
  (if (or (null? set1) (null? set2))
      '()
      (let ((x1 (car set1)) (x2 (car set2)))
        (cond ((= x1 x2)
               (cons x1 (intersection-set 
                         (cdr set1)
                         (cdr set2))))
              ((< x1 x2) (intersection-set 
                          (cdr set1) 
                          set2))
              ((< x2 x1) (intersection-set 
                          set1 
                          (cdr set2)))))))

(define (union-set set1 set2)
    (cond ((or (null? set2) (null? set1)) (append set1 set2))
            ((= (car set2) (car set1))  (cons (car set1) (union-set (cdr set1) (cdr set2))))
            ((< (car set2) (car set1)) (cons (car set2) (union-set set1 (cdr set2))))
            (else (cons (car set1) (union-set (cdr set1)  set2)))
    )

(define (list->tree elements)
  (car (partial-tree

(define (tree->list-1 tree)
  (if (null? tree)
      '()

(define (partial-tree elts n)
  (if (= n 0)
      (cons '() elts)
      (let ((left-size 
             (quotient (- n 1) 2)))
        (let ((left-result 
               (partial-tree 
                elts left-size)))
          (let ((left-tree 
                 (car left-result))

                 (cdr left-result))

                   (car non-left-elts))

                    right-size)))

                     (car right-result))

For combining the two sets into an intersection, I have to exclude all the elements that are not in both of the sets, that is I have to build a binary tree that is formed by nodes that are present in both of the inputs.

The easiest solution is to convert the trees into lists, use the intersection operation for ordered lists for them, then use list->tree to form the balanced tree.

So, let's give it a shot.

(define (intersection-set->tree set1 set2)
    (let ((list1 (tree->list-1 set1))
        (list2 (tree->list-1 set2))
        )
        (let ((intesect-list (intersection-set list1 list2)))
            (list->tree intersect-list))))

(define (union-set->tree set1 set2)
    (let ((list1 (tree->list-1 set1))
        (list2 (tree->list-1 set2))
        )
        (let ((intesect-list (union-set list1 list2)))
            (list->tree intersect-list))))
