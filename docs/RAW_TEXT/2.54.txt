this exercise asks me to define a procedure "equal?" which uses eq? on compund data with symbols.

(define (equal? a b)
      (cond  ((and (null? a) (null? b)) #t)
            ((and (pair? a) (pair? b)) (and (eq? (car a) (car b)) (equal? (cdr a) (cdr b))))
            (else (eq? a b))
      )  
)
